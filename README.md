# Folder Structure

- ahtapot-system/
  - config/            # Configuration files and settings
  - core/              # Core state management
    - actions/         # Redux or similar actions
    - reducers/        # State reducers
    - selectors/       # State selectors
  - utilities/         # Base utility functions
    - helpers.js       # General helpers
    - constants.js     # Constants used across the project
  - components/        # React or UI components
  - pages/            # Application pages or screens
  - services/         # API services
  - styles/           # Global styles and styling utilities
  - tests/            # Tests and setups
  - assets/           # Images, fonts, etc.
